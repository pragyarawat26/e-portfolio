typedef struct {

    char operation[50];

    double result;

} Calculation;

void save_to_history(Calculation calc) {

    FILE *fp = fopen("history.txt", "a");

    if (fp == NULL) {

        printf("Error opening history file.\n");

        return;

    }

    fprintf(fp, "%s = %.2lf\n", calc.operation, calc.result);

    fclose(fp);

}
void view_history() {

    FILE *fp = fopen("history.txt","r");

    if (fp == NULL) {

        printf("No history found.\n");

        return;

    }

    char ch;

    printf("\n--- Calculation History ---\n");

    while ((ch = fgetc(fp)) != EOF) {

        putchar(ch);

    }

    fclose(fp);

    printf("\n----------------------------\n");

}
double factorial(int n) {

    if (n < 0) return -1;

    double fact = 1;

    for (int I = 2; I <= n; I++) {

        fact *= I;

    }

    return fact;
}
