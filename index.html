#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <string.h>

typedef struct {
    char operation[50];
    double result;
} Calculation;

void save_to_history(Calculation calc) {
    FILE *fp = fopen("history.txt", "a");
    if (fp == NULL) {
        printf("Error opening history file.\n");
        return;
    }
    fprintf(fp, "%s = %.2lf\n", calc.operation, calc.result);
    fclose(fp);
}

void view_history() {
    FILE *fp = fopen("history.txt","r");
    if (fp == NULL) {
        printf("No history found.\n");
        return;
    }
    char ch;
    printf("\n--- Calculation History ---\n");
    while ((ch = fgetc(fp)) != EOF) {
        putchar(ch);
    }
    fclose(fp);
    printf("\n----------------------------\n");
}

double factorial(int n) {
    if (n < 0) return -1;
    double fact = 1;
    for (int I = 2; I <= n; I++) {
        fact *= I;
    }
    return fact;
}

int main() {
    int choice;
    double num1, num2, result;
    int n;
    Calculation calc;

    do {
        printf("\n--- Scientific Calculator ---\n");
        printf("1. Addition\n2. Subtraction\n3. Multiplication\n4. Division\n");
        printf("5. Sine\n6. Cosine\n7. Tangent\n8. Logarithm\n");
        printf("9. Exponentiation\n10. Factorial\n11. View History\n12. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                printf("Enter two numbers: ");
                scanf("%lf %lf", &num1, &num2);
                result = num1 + num2;
                sprintf(calc.operation, "%.2lf + %.2lf", num1, num2);
                break;
            case 2:
                printf("Enter two numbers: ");
                scanf("%lf %lf", &num1, &num2);
                result = num1 - num2;
                sprintf(calc.operation, "%.2lf - %.2lf", num1, num2);
                break;
            case 3:
                printf("Enter two numbers: ");
                scanf("%lf %lf", &num1, &num2);
                result = num1 * num2;
                sprintf(calc.operation, "%.2lf * %.2lf", num1, num2);
                break;
            case 4:
                printf("Enter dividend and divisor: ");

                if (num2 == 0) {

                result = sin(num1 * M_PI / 180);

                printf("Enter angle in degrees: ");
                scanf("%lf", &num1);
                result = cos(num1 * M_PI / 180);
                sprintf(calc.operation, "cos(%.2lf)", num1);
                break;
            case 7:
                printf("Enter angle in degrees: ");
                scanf("%lf", &num1);
                result = tan(num1 * M_PI / 180);
                sprintf(calc.operation, "tan(%.2lf)", num1);
                break;

                printf("Enter number: ");

                    printf("Error: Log undefined for non-positive numbers.\n");

                scanf("%lf %lf", &num1, &num2);

                printf("Enter a non-negative integer: ");
                scanf("%d", &n);

            default:
                printf("Invalid choice.\n");
                continue;
        }
        calc.result = result;
        printf("Result: %.2lf\n", result);
        save_to_history(calc);
    } while (1);

    return 0;
}
